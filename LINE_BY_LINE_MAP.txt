0019: // Add comma separators to the integer portion of a decimal string.
0020: fn add_commas(number: &str) -> String {
0021:     // Split a possible negative sign away from the numeric digits.
0022:     let (sign, digits) = if let Some(rest) = number.strip_prefix('-') {
0023:         // Preserve the negative sign separately.
0024:         ("-", rest)
0025:     } else {
0026:         // Use an empty sign for non-negative values.
0027:         ("", number)
0028:     };
0029: 
0030:     // Build the grouped integer from the right side in three-digit chunks.
0031:     let mut grouped = String::new();
0032:     // Read the digits in reverse order so grouping is easy.
0033:     for (index, character) in digits.chars().rev().enumerate() {
0034:         // Insert a comma before every new three-digit group.
0035:         if index > 0 && index % 3 == 0 {
0036:             // Add the thousands separator.
0037:             grouped.push(',');
0038:         }
0039:         // Add the current digit.
0040:         grouped.push(character);
0041:     }
0042: 
0043:     // Reverse the grouped string back into normal reading order.
0044:     let grouped: String = grouped.chars().rev().collect();
0045:     // Put the sign back in front of the grouped number.
0046:     format!("{sign}{grouped}")
0047: }

0049: // Format a floating-point number with two decimal places and comma separators.
0050: fn format_decimal(value: f64) -> String {
0051:     // Convert the value to exactly two decimal places.
0052:     let raw = format!("{value:.2}");
0053:     // Split the number around its decimal point.
0054:     let mut pieces = raw.split('.');
0055:     // Read the integer component.
0056:     let integer = pieces.next().unwrap_or("0");
0057:     // Read the fractional component.
0058:     let fraction = pieces.next().unwrap_or("00");
0059:     // Add grouping to the integer component.
0060:     let grouped_integer = add_commas(integer);
0061:     // Reassemble the formatted decimal number.
0062:     format!("{grouped_integer}.{fraction}")
0063: }

0002: #[derive(Clone, Debug, PartialEq, Eq, Hash)]
0003: // Store the metadata needed to interpret token amounts.
0004: pub struct Token {
0005:     // Store the human-readable ticker symbol.
0006:     pub symbol: String,
0007:     // Store the token's number of decimal places.
0008:     pub decimals: u8,
0009: }
0010: 
0011: // Implement constructors and formatting helpers for Token.
0012: impl Token {
0013:     // Create a token from a symbol and decimal count.
0014:     pub fn new(symbol: &str, decimals: u8) -> Self {
0015:         // Return a fully initialized token.
0016:         Self {
0017:             // Convert the borrowed symbol into owned string data.
0018:             symbol: symbol.to_string(),
0019:             // Store the decimal precision.
0020:             decimals,
0021:         }
0022:     }
0023: 
0024:     // Convert a raw integer amount into a human-readable decimal string.
0025:     pub fn format_raw(&self, raw_amount: u128) -> String {
0026:         // Convert the raw amount into a string so we can place the decimal point exactly.
0027:         let raw = raw_amount.to_string();
0028: 
0029:         // Handle tokens that use zero decimal places.
0030:         if self.decimals == 0 {
0031:             // Return the raw integer directly.
0032:             return raw;
0033:         }
0034: 
0035:         // Convert the decimal count to usize for string indexing.
0036:         let decimals = self.decimals as usize;
0037: 
0038:         // Add leading zeroes when the raw value has fewer digits than the decimal count.
0039:         let padded = if raw.len() <= decimals {
0040:             // Create enough leading zeroes for a value smaller than one whole token.
0041:             format!("{}{}", "0".repeat(decimals + 1 - raw.len()), raw)
0042:         } else {
0043:             // Reuse the raw string when it already has enough integer digits.
0044:             raw
0045:         };
0046: 
0047:         // Split the padded value into the integer portion and fractional portion.
0048:         let split_index = padded.len() - decimals;
0049:         // Extract the integer portion.
0050:         let integer_part = &padded[..split_index];
0051:         // Extract the fractional portion.
0052:         let fractional_part = &padded[split_index..];
0053: 
0054:         // Remove insignificant trailing zeroes from the fractional part.
0055:         let fractional_trimmed = fractional_part.trim_end_matches('0');
0056: 
0057:         // Return just the integer part when the fractional part is all zeroes.
0058:         if fractional_trimmed.is_empty() {
0059:             // Return the clean whole-token value.
0060:             return integer_part.to_string();
0061:         }
0062: 
0063:         // Combine the integer and fractional portions.
0064:         format!("{integer_part}.{fractional_trimmed}")
0065:     }
0066: }

0078:     // Check whether this pool can perform the requested token swap.
0079:     pub fn supports_pair(&self, token_in: &Token, token_out: &Token) -> bool {
0080:         // Require positive reserves because an empty pool cannot execute a quote.
0081:         let has_liquidity = self.reserve_a > 0 && self.reserve_b > 0 && self.liquidity > 0;
0082:         // Check the normal A -> B orientation.
0083:         let forward = self.token_a == *token_in && self.token_b == *token_out;
0084:         // Check the reverse B -> A orientation.
0085:         let reverse = self.token_b == *token_in && self.token_a == *token_out;
0086:         // Return true only when the pair and liquidity are both valid.
0087:         has_liquidity && (forward || reverse)
0088:     }

0090:     // Calculate a swap quote without mutating the pool.
0091:     pub fn quote(&self, token_in: &Token, amount_in: u128) -> Option<Quote> {
0092:         // Reject zero-sized swaps.
0093:         if amount_in == 0 {
0094:             // A zero-input quote has no useful economic meaning.
0095:             return None;
0096:         }
0097: 
0098:         // Determine whether token A is entering the pool.
0099:         let input_is_a = self.token_a == *token_in;
0100:         // Select the correct input reserve for the swap direction.
0101:         let reserve_in = if input_is_a {
0102:             self.reserve_a
0103:         } else {
0104:             self.reserve_b
0105:         };
0106:         // Select the correct output reserve for the swap direction.
0107:         let reserve_out = if input_is_a {
0108:             self.reserve_b
0109:         } else {
0110:             self.reserve_a
0111:         };
0112: 
0113:         // Reject empty reserves because the AMM formula would be invalid.
0114:         if reserve_in == 0 || reserve_out == 0 {
0115:             // Return no quote when the pool cannot provide liquidity.
0116:             return None;
0117:         }
0118: 
0119:         // Calculate the portion of the input that remains after the pool fee.
0120:         let fee_adjusted_input =
0121:             amount_in.saturating_mul(FEE_DENOMINATOR - self.fee_bps as u128)
0122:                 / FEE_DENOMINATOR;
0123: 
0124:         // Calculate the explicit fee in the input token's base units.
0125:         let fee_amount = amount_in.saturating_sub(fee_adjusted_input);
0126: 
0127:         // Calculate the numerator of the constant-product output formula.
0128:         let numerator = fee_adjusted_input.saturating_mul(reserve_out);
0129:         // Calculate the denominator of the constant-product output formula.
0130:         let denominator = reserve_in.saturating_add(fee_adjusted_input);
0131: 
0132:         // Reject an invalid denominator even though normal pool state makes this impossible.
0133:         if denominator == 0 {
0134:             // Return no quote when arithmetic would otherwise be invalid.
0135:             return None;
0136:         }
0137: 
0138:         // Calculate the actual output after price impact and fees.
0139:         let amount_out = numerator / denominator;
0140: 
0141:         // Calculate the theoretical output at the current spot price after fees but before price impact.
0142:         let spot_output = fee_adjusted_input.saturating_mul(reserve_out) / reserve_in;
0143: 
0144:         // Estimate price impact in basis points.
0145:         let price_impact_bps = if spot_output == 0 {
0146:             // Report zero when the theoretical output is too small to calculate impact.
0147:             0
0148:         } else {
0149:             // Calculate the percentage of theoretical output lost to price impact.
0150:             let impact_numerator = spot_output.saturating_sub(amount_out);
0151:             // Convert that ratio to basis points using integer arithmetic.
0152:             impact_numerator.saturating_mul(10_000) / spot_output
0153:         };
0154: 
0155:         // Return the complete quote object.
0156:         Some(Quote {
0157:             // Store the original input amount.
0158:             amount_in,
0159:             // Store the actual output.
0160:             amount_out,
0161:             // Store the input-token fee.
0162:             fee_amount,
0163:             // Store estimated price impact.
0164:             price_impact_bps,
0165:         })
0166:     }

0001: // Import the complete opportunity record returned from route evaluation.
0002: use super::opportunity::Opportunity;
0003: // Import the pool model used to produce swap quotes.
0004: use crate::pools::pool::Pool;
0005: // Import the token model used to define the route.
0006: use crate::tokens::token::Token;
0007: 
0008: // Define the arbitrage scanner and its configuration.
0009: pub struct ArbitrageEngine {
0010:     // Store every liquidity pool currently known to the scanner.
0011:     pools: Vec<Pool>,
0012:     // Store the minimum net profit that the caller considers actionable.
0013:     minimum_profit: u128,
0014:     // Store the estimated cost of executing both legs.
0015:     gas_cost: u128,
0016: }
0017: 
0018: // Implement market management and route evaluation.
0019: impl ArbitrageEngine {
0020:     // Create a new empty arbitrage engine.
0021:     pub fn new(minimum_profit: u128, gas_cost: u128) -> Self {
0022:         // Return a scanner with no pools loaded yet.
0023:         Self {
0024:             // Start with an empty market database.
0025:             pools: Vec::new(),
0026:             // Store the configured profit threshold.
0027:             minimum_profit,
0028:             // Store the configured gas estimate.
0029:             gas_cost,
0030:         }
0031:     }
0032: 
0033:     // Add a pool to the scanner's market database.
0034:     pub fn add_pool(&mut self, pool: Pool) {
0035:         // Append the supplied pool to the internal list.
0036:         self.pools.push(pool);
0037:     }
0038: 
0039:     // Expose the minimum-profit threshold for reporting.
0040:     pub fn minimum_profit(&self) -> u128 {
0041:         // Return the configured threshold.
0042:         self.minimum_profit
0043:     }
0044: 
0045:     // Scan all two-pool token_in -> token_mid -> token_in routes.
0046:     pub fn scan(
0047:         &self,
0048:         token_in: &Token,
0049:         token_mid: &Token,
0050:         amount_in: u128,
0051:     ) -> Vec<Opportunity> {
0052:         // Create a vector that will contain every valid route.
0053:         let mut candidates = Vec::new();
0054: 
0055:         // Try every pool as the first leg.
0056:         for first_pool in &self.pools {
0057:             // Skip pools that cannot perform token_in -> token_mid.
0058:             if !first_pool.supports_pair(token_in, token_mid) {
0059:                 // Continue to the next first-leg pool.
0060:                 continue;
0061:             }
0062: 
0063:             // Ask the first pool for a quote.
0064:             let Some(quote_first) = first_pool.quote(token_in, amount_in) else {
0065:                 // Skip this route if the first quote is invalid.
0066:                 continue;
0067:             };
0068: 
0069:             // Try every different pool as the second leg.
0070:             for second_pool in &self.pools {
0071:                 // Do not execute both legs against the same pool.
0072:                 if std::ptr::eq(first_pool, second_pool) {
0073:                     // Continue to the next second-leg pool.
0074:                     continue;
0075:                 }
0076: 
0077:                 // Skip pools that cannot perform token_mid -> token_in.
0078:                 if !second_pool.supports_pair(token_mid, token_in) {
0079:                     // Continue to the next second-leg pool.
0080:                     continue;
0081:                 }
0082: 
0083:                 // Quote the second leg using the first leg's output.
0084:                 let Some(quote_second) = second_pool.quote(token_mid, quote_first.amount_out) else {
0085:                     // Skip the route when the second quote is invalid.
0086:                     continue;
0087:                 };
0088: 
0089:                 // Save the final output returned to the starting token.
0090:                 let amount_out = quote_second.amount_out;
0091:                 // Calculate gross profit before gas.
0092:                 let gross_profit = amount_out as i128 - amount_in as i128;
0093:                 // Calculate net profit after the estimated execution cost.
0094:                 let net_profit = gross_profit - self.gas_cost as i128;
0095: 
0096:                 // Create the complete opportunity record.
0097:                 candidates.push(Opportunity {
0098:                     // Save the starting token symbol.
0099:                     token_in: token_in.symbol.clone(),
0100:                     // Save the intermediate token symbol.
0101:                     token_mid: token_mid.symbol.clone(),
0102:                     // Save starting capital.
0103:                     amount_in,
0104:                     // Save the first-leg quote.
0105:                     quote_first: quote_first.clone(),
0106:                     // Save the second-leg quote.
0107:                     quote_second,
0108:                     // Save final output.
0109:                     amount_out,
0110:                     // Save gross profit.
0111:                     gross_profit,
0112:                     // Save gas cost.
0113:                     gas_cost: self.gas_cost,
0114:                     // Save net profit.
0115:                     net_profit,
0116:                     // Store the first venue.
0117:                     buy_dex: first_pool.dex,
0118:                     // Store the second venue.
0119:                     sell_dex: second_pool.dex,
0120:                 });
0121:             }
0122:         }
0123: 
0124:         // Put the highest-net-profit route first for easy reporting.
0125:         candidates.sort_by(|left, right| right.net_profit.cmp(&left.net_profit));
0126: 
0127:         // Return every route so the caller can inspect both profitable and rejected candidates.
0128:         candidates
0129:     }
0130: 
0131: }
